# Link

A styled anchor element. Uses the primary color by default with an underline on hover. Set <code>external</code> to open
in a new tab and display a trailing external-link icon.

## Playground

<ComponentPlayground tag="lt-link" content="Visit site" defaults={{ href: 'https://example.com' }} />

## Usage

  <FrameworkTabs
    html={`<lt-link href="https://example.com">Visit site</lt-link>
<lt-link href="https://example.com" external>Open in new tab</lt-link>

<!-- target/rel without the external icon; download; underline modes -->

<lt-link href="https://example.com" target="_blank">New tab, no icon</lt-link>
<lt-link href="/report.pdf" download>Download report</lt-link>
<lt-link href="/doc" underline="always">Always underlined</lt-link>
<lt-link href="/doc" underline="none">Never underlined</lt-link>`}
    react={`

<lt-link href="/doc" underline="none">Never underlined</lt-link>

</template>
