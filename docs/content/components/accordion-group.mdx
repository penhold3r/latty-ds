# Accordion Group

<code>lt-accordion-group</code> wraps multiple <code>lt-accordion</code> children and coordinates them:
opening one closes the others (exclusive mode). Set <code>multiple</code> to let more than one stay open
at a time.

Each <code>lt-accordion</code> is a standalone disclosure widget on its own — this component adds the
cross-panel coordination. See the <a href="/components/accordion">Accordion</a> page for the full
accordion API.

## Usage

<FrameworkTabs
  html={`
<lt-accordion-group>
  <lt-accordion heading="Shipping">Standard and express options.</lt-accordion>
  <lt-accordion heading="Returns">30-day return policy.</lt-accordion>
  <lt-accordion heading="Warranty">1-year limited warranty.</lt-accordion>
</lt-accordion-group>

<!-- Allow multiple panels open at once -->
<lt-accordion-group multiple>
  <lt-accordion heading="Step 1">First step details.</lt-accordion>
  <lt-accordion heading="Step 2">Second step details.</lt-accordion>
</lt-accordion-group>
  `}
  react={`
import { AccordionGroup, Accordion } from '@latty-ds/react';

<AccordionGroup>
  <Accordion heading="Shipping">Standard and express options.</Accordion>
  <Accordion heading="Returns">30-day return policy.</Accordion>
  <Accordion heading="Warranty">1-year limited warranty.</Accordion>
</AccordionGroup>
  `}
  vue={`
<template>
  <lt-accordion-group>
    <lt-accordion heading="Shipping">Standard and express options.</lt-accordion>
    <lt-accordion heading="Returns">30-day return policy.</lt-accordion>
    <lt-accordion heading="Warranty">1-year limited warranty.</lt-accordion>
  </lt-accordion-group>
</template>
  `}
/>
