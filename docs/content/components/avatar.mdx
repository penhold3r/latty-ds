# Avatar

Displays a user avatar. Falls back to initials derived from <code>name</code>, then to a generic person icon if
neither is available.

## Playground

<ComponentPlayground tag="lt-avatar" defaults={{ name: 'Jane Doe' }} />

## Usage

  <FrameworkTabs
    html={`
<lt-avatar src="https://i.pravatar.cc/150" name="Jane Doe"></lt-avatar>
<lt-avatar name="Jane Doe" size="lg"></lt-avatar>
<lt-avatar name="Jane Doe" shape="square"></lt-avatar>
<lt-avatar size="md"></lt-avatar>

<!-- Presence status dot -->

<lt-avatar name="Jane Doe" status="online"></lt-avatar>
<lt-avatar name="Jane Doe" status="busy"></lt-avatar>
`}
    react={`
import { Avatar } from '@latty-ds/react';

<Avatar src="https://i.pravatar.cc/150" name="Jane Doe" />
<Avatar name="Jane Doe" size="lg" />
<Avatar name="Jane Doe" shape="square" />
<Avatar size="md" />

// Presence status dot

<Avatar name="Jane Doe" status="online" />
<Avatar name="Jane Doe" status="busy" />
    `}
    vue={`
<template>
  <lt-avatar src="https://i.pravatar.cc/150" name="Jane Doe"></lt-avatar>
  <lt-avatar name="Jane Doe" size="lg"></lt-avatar>
  <lt-avatar name="Jane Doe" shape="square"></lt-avatar>
  <lt-avatar size="md"></lt-avatar>

  <!-- Presence status dot -->

<lt-avatar name="Jane Doe" status="online"></lt-avatar>
<lt-avatar name="Jane Doe" status="busy"></lt-avatar>

</template>
    `}
  />
