# Radio

Radio buttons for single-selection within a group. Use <code>lt-radio-group</code> to manage a named group with a shared
label, value, and validation state.

## Playground

## Usage

  <FrameworkTabs
    html={`
<lt-radio-group label="Preferred contact" name="contact" value="email">
  <lt-radio label="Email" value="email"></lt-radio>
  <lt-radio label="Phone" value="phone"></lt-radio>
  <lt-radio label="Post" value="post" disabled></lt-radio>
</lt-radio-group>
    `}
    react={`
import { RadioGroup, Radio } from '@latty-ds/react';

<RadioGroup label="Preferred contact" name="contact" value="email">
  <Radio label="Email" value="email" />
  <Radio label="Phone" value="phone" />
  <Radio label="Post" value="post" disabled />
</RadioGroup>
    `}
    vue={`
<template>
  <lt-radio-group label="Preferred contact" name="contact" value="email">
    <lt-radio label="Email" value="email"></lt-radio>
    <lt-radio label="Phone" value="phone"></lt-radio>
    <lt-radio label="Post" value="post" disabled></lt-radio>
  </lt-radio-group>
</template>
    `}
  />
