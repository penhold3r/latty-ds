# Calendar

An always-visible date-selection grid. Supports single-date and date-range selection, locale-aware day and month
names, configurable week start, min/max constraints, and full keyboard navigation.

## Playground

<ComponentPlayground
  tag="lt-calendar"
  defaults={{ locale: 'en-US', 'week-start': '0', 'show-outside-days': true, mode: 'single' }}
/>

## Usage

  <FrameworkTabs
    html={`
<!-- Basic -->
<lt-calendar></lt-calendar>

<!-- Pre-selected date -->

<lt-calendar value="2026-06-15"></lt-calendar>

<!-- Date range selection -->

<lt-calendar mode="range" value-start="2026-06-01" value-end="2026-06-15"></lt-calendar>

<!-- Two-month view -->

<lt-calendar months="2"></lt-calendar>

<!-- Two-month range picker -->

<lt-calendar mode="range" months="2"></lt-calendar>

<!-- With min/max range -->

<lt-calendar min="2026-01-01" max="2026-12-31"></lt-calendar>

<!-- Monday week start, French locale -->

<lt-calendar week-start="1" locale="fr-FR"></lt-calendar>
`}
    react={`
import { Calendar } from '@latty-ds/react';

// Basic

<Calendar />

// Pre-selected date

<Calendar value="2026-06-15" />

// Date range selection

<Calendar mode="range" valueStart="2026-06-01" valueEnd="2026-06-15" />

// Two-month view

<Calendar months={2} />

// Two-month range picker

<Calendar mode="range" months={2} />

// With min/max range

<Calendar min="2026-01-01" max="2026-12-31" />

// Monday week start, French locale

<Calendar weekStart="1" locale="fr-FR" />
    `}
    vue={`
<template>

<lt-calendar :months="2"></lt-calendar>

  <!-- Two-month range picker -->

<lt-calendar mode="range" :months="2"></lt-calendar>

  <!-- With min/max range -->

<lt-calendar min="2026-01-01" max="2026-12-31"></lt-calendar>

  <!-- Monday week start, French locale -->

<lt-calendar week-start="1" locale="fr-FR"></lt-calendar>

</template>
    `}
  />
